cmake_minimum_required(VERSION 3.25)
project(KantanLogoMaker VERSION 0.1.0 LANGUAGES C CXX)
option(KANTAN_BUILD_APP "Build the Qt desktop application" ON)
option(KANTAN_BUILD_TESTS "Build headless contract tests" ON)
option(KANTAN_SANITIZE "Instrument owned code for memory errors" OFF)
# The release build pins Qt 6.8.3 exactly. Other hosts (e.g. a Linux validation build against a distro Qt)
# may relax this; the Windows presets never do.
set(KANTAN_QT_VERSION "6.8.3" CACHE STRING "Required Qt version")
option(KANTAN_QT_EXACT "Require exactly KANTAN_QT_VERSION" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Owned-code warnings. GCC/Clang add -Wconversion so narrowing is caught as MSVC /W4 would.
if(MSVC)
  # Vendored headers are included with angle brackets and treated as external (no warnings).
  set(KANTAN_WARNINGS /W4 /WX /permissive- /utf-8 /external:anglebrackets /external:W0)
else()
  set(KANTAN_WARNINGS -Wall -Wextra -Wpedantic -Wconversion -Wno-missing-field-initializers -Werror)
endif()

# Pinned, vendored source-only dependencies (see dependencies.json). Their warnings stay isolated.
add_library(clipper2 STATIC
  third_party/clipper2/src/clipper.engine.cpp third_party/clipper2/src/clipper.offset.cpp
  third_party/clipper2/src/clipper.rectclip.cpp)
target_include_directories(clipper2 SYSTEM PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/clipper2/include)
if(MSVC)
  target_compile_options(clipper2 PRIVATE /W0 /utf-8)
else()
  target_compile_options(clipper2 PRIVATE -w)
endif()

add_library(kantan_core
  core/geometry/Geometry.cpp core/raster/Rasterizer.cpp core/composite/Composite.cpp core/shapes/Shapes.cpp
  core/boolean/Boolean.cpp core/document/Document.cpp core/effects/Effects.cpp core/render/Render.cpp
  core/render/ImageSampling.cpp core/editing/Editor.cpp core/editing/Transforms.cpp core/editing/Colors.cpp)
target_include_directories(kantan_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core)
target_link_libraries(kantan_core PRIVATE clipper2)
target_compile_options(kantan_core PRIVATE ${KANTAN_WARNINGS})
if(KANTAN_SANITIZE)
  if(MSVC)
    target_compile_options(kantan_core PUBLIC /fsanitize=address)
    target_link_options(kantan_core PUBLIC /INCREMENTAL:NO)
  else()
    target_compile_options(kantan_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      # GCC 12/13 report false array-bounds errors for one-element initializer lists under sanitizers.
      target_compile_options(kantan_core PUBLIC -Wno-array-bounds -Wno-stringop-overread -Wno-stringop-overflow)
    endif()
    target_link_options(kantan_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()

include(CTest)
if(KANTAN_BUILD_TESTS)
  add_executable(kantan_core_tests tests/core_tests.cpp)
  target_link_libraries(kantan_core_tests PRIVATE kantan_core)
  target_compile_options(kantan_core_tests PRIVATE ${KANTAN_WARNINGS})
  add_test(NAME core_contracts COMMAND kantan_core_tests)
  add_executable(kantan_render_tests tests/render_tests.cpp)
  target_link_libraries(kantan_render_tests PRIVATE kantan_core)
  target_compile_options(kantan_render_tests PRIVATE ${KANTAN_WARNINGS})
  add_test(NAME render_contracts COMMAND kantan_render_tests)
  if(KANTAN_SANITIZE AND MSVC)
    get_filename_component(COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(core_contracts render_contracts PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${COMPILER_BIN}")
  endif()
  add_test(NAME architecture COMMAND ${CMAKE_COMMAND} -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckArchitecture.cmake)
  add_executable(kantan_bench benchmarks/render_bench.cpp)
  target_link_libraries(kantan_bench PRIVATE kantan_core)
endif()

if(KANTAN_BUILD_APP)
  include(cmake/FindProjectQt.cmake)
  if(KANTAN_QT_EXACT)
    find_package(Qt6 ${KANTAN_QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Widgets Concurrent Test)
  else()
    find_package(Qt6 ${KANTAN_QT_VERSION} REQUIRED COMPONENTS Core Gui Widgets Concurrent Test)
  endif()
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  set(MINIZ_ROOT "${CMAKE_SOURCE_DIR}/.deps/miniz-3.1.0" CACHE PATH "Pinned Miniz source")
  add_library(miniz STATIC ${MINIZ_ROOT}/miniz.c ${MINIZ_ROOT}/miniz_tdef.c ${MINIZ_ROOT}/miniz_tinfl.c ${MINIZ_ROOT}/miniz_zip.c)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/miniz_export.h" "#pragma once\n#define MINIZ_EXPORT\n")
  target_include_directories(miniz SYSTEM PUBLIC ${MINIZ_ROOT} ${CMAKE_CURRENT_BINARY_DIR})
  target_compile_definitions(miniz PUBLIC MINIZ_NO_STDIO MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
  if(NOT MSVC)
    target_compile_options(miniz PRIVATE -w)
  endif()
  add_library(nanosvg STATIC third_party/nanosvg/nanosvg_impl.c)
  target_include_directories(nanosvg SYSTEM PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/nanosvg)
  if(MSVC)
    target_compile_options(nanosvg PRIVATE /W0)
    target_compile_definitions(nanosvg PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(nanosvg PRIVATE -w)
  endif()
  add_library(kantan_io io/ProjectIO.cpp io/Serialization.cpp io/Images.cpp io/WriteCoordinator.cpp io/IcoWriter.cpp io/ExportJob.cpp
    io/TextShaper.cpp io/FontLibrary.cpp io/SvgImport.cpp io/ExportPresets.cpp)
  target_include_directories(kantan_io PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(kantan_io PUBLIC kantan_core Qt6::Core Qt6::Gui PRIVATE miniz nanosvg)
  set(KANTAN_APP_SOURCES app/MainWindow.cpp app/CanvasView.cpp app/ObjectPanel.cpp app/Inspector.cpp
    app/PresetPanel.cpp app/SizeStrip.cpp app/ExportDialog.cpp app/ArtboardDialog.cpp app/ToolIcons.cpp
    app/Presets.cpp app/ColorField.cpp app/PalettePanel.cpp app/MockupDialog.cpp app/RecentFiles.cpp app/StartDialog.cpp app/icons.qrc app/presets.qrc)
  add_executable(kantan_app WIN32 app/main.cpp ${KANTAN_APP_SOURCES})
  set_target_properties(kantan_app PROPERTIES OUTPUT_NAME KantanLogoMaker)
  target_link_libraries(kantan_app PRIVATE kantan_core kantan_io Qt6::Widgets Qt6::Concurrent)
  if(MSVC)
    target_compile_options(kantan_io PRIVATE /W4 /WX /utf-8 /external:anglebrackets /external:W0)
    target_compile_options(kantan_app PRIVATE /W4 /WX /utf-8 /external:anglebrackets /external:W0)
  else()
    target_compile_options(kantan_io PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Werror)
    target_compile_options(kantan_app PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Werror)
  endif()
  if(KANTAN_BUILD_TESTS)
    add_executable(kantan_io_tests tests/io_tests.cpp)
    target_compile_definitions(kantan_io_tests PRIVATE KANTAN_FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
    target_link_libraries(kantan_io_tests PRIVATE kantan_io miniz)
    add_test(NAME io_contracts COMMAND kantan_io_tests)
    add_executable(kantan_ui_tests tests/ui_tests.cpp ${KANTAN_APP_SOURCES})
    target_compile_definitions(kantan_ui_tests PRIVATE KANTAN_FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
    target_link_libraries(kantan_ui_tests PRIVATE kantan_io Qt6::Widgets Qt6::Concurrent Qt6::Test)
    add_test(NAME ui_contracts COMMAND kantan_ui_tests -platform offscreen -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/ui-contracts.xml,junitxml")
    set_tests_properties(ui_contracts PROPERTIES TIMEOUT 90)
    if(WIN32)
      file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-temp")
      get_filename_component(KANTAN_QT_ROOT "${Qt6_DIR}/../../.." ABSOLUTE)
      set_tests_properties(io_contracts ui_contracts PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${KANTAN_QT_ROOT}/bin;TMP=set:${CMAKE_CURRENT_BINARY_DIR}/test-temp;TEMP=set:${CMAKE_CURRENT_BINARY_DIR}/test-temp" ENVIRONMENT "QT_QPA_PLATFORM_PLUGIN_PATH=${KANTAN_QT_ROOT}/plugins/platforms")
    endif()
    set_property(TEST ui_contracts APPEND PROPERTY ENVIRONMENT "QT_QPA_FONTDIR=${CMAKE_SOURCE_DIR}/tests/fixtures")
  endif()
  install(TARGETS kantan_app RUNTIME DESTINATION .)
  if(MSVC)
    # A packaged app must not depend on Visual Studio being installed on the user's machine.
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ".")
    include(InstallRequiredSystemLibraries)
  endif()
endif()
